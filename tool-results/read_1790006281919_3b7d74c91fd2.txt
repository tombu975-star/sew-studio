   104→function roleLabel(role: string): string {
   105→  if (role === "madam") return "Madam";
   106→  if (!role) return "Staff";
   107→  return role.charAt(0).toUpperCase() + role.slice(1);
   108→}
   109→
   110→function initialOf(name: string): string {
   111→  return (name || "?").trim().charAt(0).toUpperCase() || "?";
   112→}

   284→/* ======================= Staff & Trainers card ======================= */
   285→
   286→function StaffTrainersCard({
   287→  users,
   288→  isLoading,
   289→  ownerId,
   290→  onAdd,
   291→}: {
   292→  users: User[];
   293→  isLoading: boolean;
   294→  ownerId?: string;
   295→  onAdd: () => void;
   296→}) {
   297→  const sorted = useMemo(() => {
   298→    return [...users].sort((a, b) => {
   299→      const ao = ROLE_ORDER[a.role] ?? 9;
   300→      const bo = ROLE_ORDER[b.role] ?? 9;
   301→      if (ao !== bo) return ao - bo;
   302→      return a.fullName.localeCompare(b.fullName);
   303→    });
   304→  }, [users]);
   305→
   306→  return (
   307→    <SectionCard
   308→      title={`Atelier Roster · ${users.length}`}
   309→      bodyClassName="p-0"
   310→      action={
   311→        <PremiumButton size="sm" onClick={onAdd}>
   312→          <UserPlus className="w-4 h-4" /> Add Member
   313→        </PremiumButton>
   314→      }
   315→    >
   316→      {isLoading ? (
   317→        <SkeletonList rows={4} />
   318→      ) : sorted.length === 0 ? (
   319→        <EmptyState
   320→          icon={UsersRound}
   321→          title="No members yet"
   322→          description="Add your first staff member or trainer to start building the atelier roster."
   323→          action={
   324→            <PremiumButton size="sm" onClick={onAdd}>
   325→              <UserPlus className="w-4 h-4" /> Add Member
   326→            </PremiumButton>
   327→          }
   328→        />
   329→      ) : (
   330→        <div className="divide-y divide-[#E7E5E4]/70">
   331→          {sorted.map((u) => {
   332→            const isOwner = u.role === "madam";
   333→            const isMe = ownerId === u.id;
   334→            return (
   335→              <div
   336→                key={u.id}
   337→                className="group flex items-center gap-4 px-5 py-4 hover:bg-[#faf8f5] transition-colors"
   338→              >
   339→                <div
   340→                  className="w-11 h-11 rounded-full bg-gradient-to-br from-[#800020] to-[#6b0018] flex items-center justify-center text-white font-display font-semibold text-lg shrink-0 shadow-sm"
   341→                  aria-hidden
   342→                >
   343→                  {initialOf(u.fullName)}
   344→                </div>
   345→                <div className="flex-1 min-w-0">
   346→                  <div className="flex items-center gap-2 flex-wrap">
   347→                    <span className="font-medium text-[#1C1917] truncate">
   348→                      {u.fullName}
   349→                    </span>
   350→                    {isOwner && (
   351→                      <span className="inline-flex items-center gap-1 text-[10px] font-semibold uppercase tracking-[0.16em] text-[#800020] bg-[#800020]/8 border border-[#800020]/15 px-1.5 py-0.5 rounded-md">
   352→                        <Crown className="w-3 h-3" /> Owner
   353→                      </span>
   354→                    )}
   355→                    {isMe && !isOwner && (
   356→                      <span className="inline-flex items-center text-[10px] font-medium uppercase tracking-[0.18em] text-stone-500 bg-stone-100 px-1.5 py-0.5 rounded-md">
   357→                        You
   358→                      </span>
   359→                    )}
   360→                  </div>
   361→                  <div className="flex flex-wrap items-center gap-x-4 gap-y-1 text-xs text-stone-500 mt-1">
   362→                    {u.email && (
   363→                      <span className="inline-flex items-center gap-1 truncate max-w-[220px]">
   364→                        <Mail className="w-3 h-3 shrink-0" /> {u.email}
   365→                      </span>
   366→                    )}
   367→                    {u.phone && (
   368→                      <span className="inline-flex items-center gap-1">
   369→                        <Phone className="w-3 h-3" /> {u.phone}
   370→                      </span>
   371→                    )}
   372→                    {u.gender && (
   373→                      <span className="inline-flex items-center rounded-md bg-[#FBF9F7] border border-[#E7E5E4] px-1.5 py-0.5 text-[10px] uppercase tracking-[0.14em] text-stone-500">
   374→                        {u.gender}
   375→                      </span>
   376→                    )}
   377→                    {!u.email && !u.phone && !u.gender && (
   378→                      <span className="italic text-stone-400">
   379→                        No contact details
   380→                      </span>
   381→                    )}
   382→                  </div>
   383→                </div>
   384→                <div className="flex items-center gap-2 shrink-0">
   385→                  <StatusBadge status={roleLabel(u.role)} />
   386→                  {isOwner && (
   387→                    <span className="hidden sm:inline text-[10px] uppercase tracking-[0.18em] text-stone-300">
   388→                      Permanent
   389→                    </span>
   390→                  )}
   391→                </div>
   392→              </div>
   393→            );
   394→          })}
   395→        </div>
   396→      )}
   397→    </SectionCard>
   398→  );
   399→}

   401→/* ========================== Freelancers card ========================= */
   402→
   403→function FreelancersCard({
   404→  freelancers,
   405→  isLoading,
   406→  onAdd,
   407→}: {
   408→  freelancers: Freelancer[];
   409→  isLoading: boolean;
   410→  onAdd: () => void;
   411→}) {
   412→  const qc = useQueryClient();
   413→  const [deleteId, setDeleteId] = useState<string | null>(null);
   414→
   415→  const statusMutation = useMutation({
   416→    mutationFn: ({ id, status }: { id: string; status: string }) =>
   417→      apiPatch<Freelancer>(`/api/freelancers/${id}`, { status }),
   418→    onSuccess: (f) => {
   419→      qc.invalidateQueries({ queryKey: ["freelancers"] });
   420→      toast.success(`${f.name} is now ${f.status}`);
   421→    },
   422→    onError: () => toast.error("Could not update status."),
   423→  });
   424→
   425→  const deleteMutation = useMutation({
   426→    mutationFn: (id: string) => apiDelete(`/api/freelancers/${id}`),
   427→    onSuccess: () => {
   428→      qc.invalidateQueries({ queryKey: ["freelancers"] });
   429→      toast.success("Freelancer removed from roster.");
   430→      setDeleteId(null);
   431→    },
   432→    onError: () => toast.error("Could not remove freelancer."),
   433→  });
   434→
   435→  const target = freelancers.find((f) => f.id === deleteId) || null;
   436→
   437→  return (
   438→    <SectionCard
   439→      title={`Commissioned Talent · ${freelancers.length}`}
   440→      bodyClassName="p-0"
   441→      action={
   442→        <PremiumButton size="sm" onClick={onAdd}>
   443→          <Plus className="w-4 h-4" /> Add Freelancer
   444→        </PremiumButton>
   445→      }
   446→    >
   447→      {isLoading ? (
   448→        <SkeletonList rows={4} />
   449→      ) : freelancers.length === 0 ? (
   450→        <EmptyState
   451→          icon={UsersRound}
   452→          title="No freelancers yet"
   453→          description="Add the beaders, embroiderers, cutters and tailors you commission for bespoke work."
   454→          action={
   455→            <PremiumButton size="sm" onClick={onAdd}>
   456→              <Plus className="w-4 h-4" /> Add Freelancer
   457→            </PremiumButton>
   458→          }
   459→        />
   460→      ) : (
   461→        <div className="divide-y divide-[#E7E5E4]/70">
   462→          {freelancers.map((f) => {
   463→            const SpecialtyIcon =
   464→              SPECIALTY_ICON[f.specialty || "Other"] || Briefcase;
   465→            return (
   466→              <div
   467→                key={f.id}
   468→                className="group flex flex-col sm:flex-row sm:items-center gap-3 sm:gap-4 px-5 py-4 hover:bg-[#faf8f5] transition-colors"
   469→              >
   470→                <div
   471→                  className="w-11 h-11 rounded-full bg-gradient-to-br from-[#800020] to-[#6b0018] flex items-center justify-center text-white shrink-0 shadow-sm"
   472→                  aria-hidden
   473→                >
   474→                  <SpecialtyIcon className="w-5 h-5" />
   475→                </div>
   476→                <div className="flex-1 min-w-0">
   477→                  <div className="flex items-center gap-2 flex-wrap">
   478→                    <span className="font-medium text-[#1C1917] truncate">
   479→                      {f.name}
   480→                    </span>
   481→                    {f.specialty && (
   482→                      <span className="inline-flex items-center rounded-md bg-[#800020]/8 text-[#800020] px-1.5 py-0.5 text-[10px] font-semibold uppercase tracking-[0.14em] border border-[#800020]/15">
   483→                        {f.specialty}
   484→                      </span>
   485→                    )}
   486→                  </div>
   487→                  <div className="flex flex-wrap items-center gap-x-4 gap-y-1 text-xs text-stone-500 mt-1">
   488→                    {f.skills && (
   489→                      <span className="italic text-stone-400 truncate max-w-[280px]">
   490→                        {f.skills}
   491→                      </span>
   492→                    )}
   493→                    {f.phone && (
   494→                      <span className="inline-flex items-center gap-1">
   495→                        <Phone className="w-3 h-3" /> {f.phone}
   496→                      </span>
   497→                    )}
   498→                    {f.email && (
   499→                      <span className="inline-flex items-center gap-1 truncate max-w-[200px]">
   500→                        <Mail className="w-3 h-3 shrink-0" /> {f.email}
   501→                      </span>
   502→                    )}
   503→                    {!f.skills && !f.phone && !f.email && (
   504→                      <span className="italic text-stone-400">
   505→                        No contact details
   506→                      </span>
   507→                    )}
   508→                  </div>
   509→                </div>
   510→                <div className="flex items-center justify-between sm:justify-end gap-3 sm:gap-4 shrink-0 pl-14 sm:pl-0">
   511→                  <div className="text-right">
   512→                    <div className="font-display font-semibold text-[#1C1917] tabular-nums text-sm leading-none">
   513→                      {f.rate ? formatCurrency(f.rate) : "—"}
   514→                    </div>
   515→                    <div className="text-[10px] uppercase tracking-[0.14em] text-stone-400 mt-1">
   516→                      per piece
   517→                    </div>
   518→                  </div>
   519→                  <StatusBadge status={f.status} />
   520→                  <Select
   521→                    value={f.status}
   522→                    onValueChange={(v) =>
   523→                      statusMutation.mutate({ id: f.id, status: v })
   524→                    }
   525→                  >
   526→                    <SelectTrigger
   527→                      size="sm"
   528→                      className="w-[130px] bg-white border-[#E7E5E4] rounded-md h-9"
   529→                      disabled={statusMutation.isPending}
   530→                      aria-label={`Change status for ${f.name}`}
   531→                    >
   532→                      <SelectValue />
   533→                    </SelectTrigger>
   534→                    <SelectContent>
   535→                      {FREELANCER_STATUSES.map((s) => (
   536→                        <SelectItem key={s} value={s}>
   537→                          {s}
   538→                        </SelectItem>
   539→                      ))}
   540→                    </SelectContent>
   541→                  </Select>
   542→                  <PremiumButton
   543→                    variant="ghost"
   544→                    size="icon"
   545→                    onClick={() => setDeleteId(f.id)}
   546→                    aria-label={`Remove ${f.name}`}
   547→                    className="text-stone-400 hover:text-[#b91c1c] hover:bg-[#b91c1c]/8"
   548→                  >
   549→                    <Trash2 className="w-4 h-4" />
   550→                  </PremiumButton>
   551→                </div>
   552→              </div>
   553→            );
   554→          })}
   555→        </div>
   556→      )}
   557→
   558→      <AlertDialog open={!!deleteId} onOpenChange={(o) => !o && setDeleteId(null)}>
   559→        <AlertDialogContent>
   560→          <AlertDialogHeader>
   561→            <AlertDialogTitle>Remove freelancer?</AlertDialogTitle>
   562→            <AlertDialogDescription>
   563→              {target
   564→                ? `This will permanently remove ${target.name} from your roster. This action cannot be undone.`
   565→                : "This will permanently remove the freelancer from your roster. This action cannot be undone."}
   566→            </AlertDialogDescription>
   567→          </AlertDialogHeader>
