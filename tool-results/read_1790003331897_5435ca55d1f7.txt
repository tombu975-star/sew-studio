    68→function initials(name: string) {
    69→  return name
    70→    .split(" ")
    71→    .filter(Boolean)
    72→    .slice(0, 2)
    73→    .map((w) => w[0]?.toUpperCase())
    74→    .join("");
    75→}
    76→
    77→function hashIndex(s: string, mod: number) {
    78→  let h = 0;
    79→  for (let i = 0; i < s.length; i++) h = (h * 31 + s.charCodeAt(i)) >>> 0;
    80→  return h % mod;
    81→}

   645→function Stat({
   646→  label,
   647→  value,
   648→  primary,
   649→  tone,
   650→  sub,
   651→}: {
   652→  label: string;
   653→  value: string;
   654→  primary?: boolean;
   655→  tone?: "ok" | "warn";
   656→  sub?: string;
   657→}) {
   658→  return (
   659→    <div>
   660→      <div className="text-[10px] uppercase tracking-[0.18em] text-stone-500 font-semibold">
   661→        {label}
   662→      </div>
   663→      <div
   664→        className={cn(
   665→          "mt-1 text-lg font-display font-semibold tabular-nums",
   666→          primary && "text-[#1C1917]",
   667→          tone === "ok" && "text-emerald-700",
   668→          tone === "warn" && "text-amber-700",
   669→          !primary && !tone && "text-[#1C1917]"
   670→        )}
   671→      >
   672→        {value}
   673→      </div>
   674→      {sub && <div className="text-[10px] text-stone-400 mt-0.5">{sub}</div>}
   675→    </div>
   676→  );
   677→}

   689→function NotesEditor({
   690→  initial,
   691→  onSave,
   692→  saving,
   693→}: {
   694→  initial: string;
   695→  onSave: (v: string) => void;
   696→  saving: boolean;
   697→}) {
   698→  const [val, setVal] = useState(initial);
   699→  const [editing, setEditing] = useState(false);
   700→  const [lastInitial, setLastInitial] = useState(initial);
   701→  if (initial !== lastInitial) {
   702→    setLastInitial(initial);
   703→    setVal(initial);
   704→  }
   705→
   706→  if (!editing) {
   707→    return (
   708→      <div>
   709→        {initial ? (
   710→          <p className="text-sm text-stone-700 whitespace-pre-wrap leading-relaxed">
   711→            {initial}
   712→          </p>
   713→        ) : (
   714→          <p className="text-sm text-stone-400">No notes yet.</p>
   715→        )}
   716→        <div className="mt-3">
   717→          <PremiumButton
   718→            variant="outline"
   719→            size="sm"
   720→            onClick={() => setEditing(true)}
   721→          >
   722→            <StickyNote className="w-3.5 h-3.5" /> Edit notes
   723→          </PremiumButton>
   724→        </div>
   725→      </div>
   726→    );
   727→  }
   728→  return (
   729→    <div>
   730→      <Textarea
   731→        value={val}
   732→        onChange={(e) => setVal(e.target.value)}
   733→        rows={5}
   734→        placeholder="Notes about this order…"
   735→        className="rounded-md"
   736→      />
   737→      <div className="flex gap-2 mt-3">
   738→        <PremiumButton
   739→          size="sm"
   740→          loading={saving}
   741→          disabled={saving}
   742→          onClick={() => {
   743→            onSave(val);
   744→            setEditing(false);
   745→          }}
   746→        >
   747→          Save
   748→        </PremiumButton>
   749→        <PremiumButton
   750→          variant="outline"
   751→          size="sm"
   752→          onClick={() => {
   753→            setVal(initial);
   754→            setEditing(false);
   755→          }}
   756→        >
   757→          Cancel
   758→        </PremiumButton>
   759→      </div>
   760→    </div>
   761→  );
   762→}
