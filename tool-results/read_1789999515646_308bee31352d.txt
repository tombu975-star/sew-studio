   281→### Time-ago helper (inline in Community)
   282→```
   283→function timeAgo(iso: string) {
   284→  const d = new Date(iso).getTime();
   285→  const diff = Date.now() - d;
   286→  const m = Math.floor(diff / 60000);
   287→  if (m < 1) return "just now";
   288→  if (m < 60) return `${m}m ago`;
   289→  const h = Math.floor(m / 60);
   290→  if (h < 24) return `${h}h ago`;
   291→  const days = Math.floor(h / 24);
   292→  if (days < 7) return `${days}d ago`;
   293→  return new Date(iso).toLocaleDateString("en-GB", { day: "numeric", month: "short" });
   294→}
   295→```
