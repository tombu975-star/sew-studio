    64→function orderPaidTotal(orderId: string, payments: Payment[]) {
    65→  return payments
    66→    .filter((p) => p.orderId === orderId)
    67→    .reduce((s, p) => s + (Number(p.amount) || 0), 0);
    68→}
    69→
    70→function orderCostTotal(order: Order) {
    71→  return (Number(order.fabricCost) || 0) + (Number(order.labourCost) || 0);
    72→}
    73→
    74→function sameMonth(d: string | Date | null | undefined, ref: Date) {
    75→  if (!d) return false;
    76→  const x = new Date(d);
    77→  if (isNaN(x.getTime())) return false;
    78→  return x.getMonth() === ref.getMonth() && x.getFullYear() === ref.getFullYear();
    79→}
    80→
    81→/** Build last `months` monthly income buckets from a flat payments array. */
    82→function monthlyIncome(payments: Payment[], months = 6) {
    83→  const now = new Date();
    84→  const buckets: { key: string; label: string; total: number }[] = [];
    85→  for (let i = months - 1; i >= 0; i--) {
    86→    const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
    87→    buckets.push({
    88→      key: `${d.getFullYear()}-${d.getMonth()}`,
    89→      label: d.toLocaleDateString("en-GB", { month: "short" }),
    90→      total: 0,
    91→    });
    92→  }
    93→  for (const p of payments) {
    94→    if (!p.date) continue;
    95→    const d = new Date(p.date);
    96→    if (isNaN(d.getTime())) continue;
    97→    const key = `${d.getFullYear()}-${d.getMonth()}`;
    98→    const b = buckets.find((x) => x.key === key);
    99→    if (b) b.total += Number(p.amount) || 0;
   100→  }
   101→  return buckets;
   102→}
