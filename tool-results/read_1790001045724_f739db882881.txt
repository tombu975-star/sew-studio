   242→/* ======================= Staff & Trainers card ======================= */
   243→
   244→function StaffTrainersCard({
   245→  users,
   246→  isLoading,
   247→  ownerId,
   248→  onAdd,
   249→}: {
   250→  users: User[];
   251→  isLoading: boolean;
   252→  ownerId?: string;
   253→  onAdd: () => void;
   254→}) {
   255→  const sorted = useMemo(() => {
   256→    return [...users].sort((a, b) => {
   257→      const ao = ROLE_ORDER[a.role] ?? 9;
   258→      const bo = ROLE_ORDER[b.role] ?? 9;
   259→      if (ao !== bo) return ao - bo;
   260→      return a.fullName.localeCompare(b.fullName);
   261→    });
   262→  }, [users]);
   263→
   264→  return (
   265→    <SectionCard
   266→      title={`Atelier Roster · ${users.length}`}
   267→      bodyClassName="p-0"
   268→      action={
   269→        <PremiumButton size="sm" onClick={onAdd}>
   270→          <UserPlus className="w-4 h-4" /> Add Member
   271→        </PremiumButton>
   272→      }
   273→    >
   274→      {isLoading ? (
   275→        <SkeletonList rows={4} />
   276→      ) : sorted.length === 0 ? (
   277→        <EmptyState
   278→          icon={UsersRound}
   279→          title="No members yet"
   280→          description="Add your first staff member or trainer to start building the atelier roster."
   281→          action={
   282→            <PremiumButton size="sm" onClick={onAdd}>
   283→              <UserPlus className="w-4 h-4" /> Add Member
   284→            </PremiumButton>
   285→          }
   286→        />
   287→      ) : (
   288→        <div className="divide-y divide-[#E7E5E4]/70">
   289→          {sorted.map((u) => {
   290→            const isOwner = u.role === "madam";
   291→            const isMe = ownerId === u.id;
   292→            return (
   293→              <div
   294→                key={u.id}
   295→                className="group flex items-center gap-4 px-5 py-4 hover:bg-[#faf8f5] transition-colors"
   296→              >
   297→                <div
   298→                  className="w-11 h-11 rounded-full bg-gradient-to-br from-[#800020] to-[#6b0018] flex items-center justify-center text-white font-display font-semibold text-lg shrink-0 shadow-sm"
   299→                  aria-hidden
   300→                >
   301→                  {initialOf(u.fullName)}
   302→                </div>
   303→                <div className="flex-1 min-w-0">
   304→                  <div className="flex items-center gap-2 flex-wrap">
   305→                    <span className="font-medium text-[#1C1917] truncate">
   306→                      {u.fullName}
   307→                    </span>
   308→                    {isOwner && (
   309→                      <span className="inline-flex items-center gap-1 text-[10px] font-semibold uppercase tracking-[0.16em] text-[#800020] bg-[#800020]/8 border border-[#800020]/15 px-1.5 py-0.5 rounded-md">
   310→                        <Crown className="w-3 h-3" /> Owner
   311→                      </span>
   312→                    )}
   313→                    {isMe && !isOwner && (
   314→                      <span className="inline-flex items-center text-[10px] font-medium uppercase tracking-[0.18em] text-stone-500 bg-stone-100 px-1.5 py-0.5 rounded-md">
   315→                        You
   316→                      </span>
   317→                    )}
   318→                  </div>
   319→                  <div className="flex flex-wrap items-center gap-x-4 gap-y-1 text-xs text-stone-500 mt-1">
   320→                    {u.email && (
   321→                      <span className="inline-flex items-center gap-1 truncate max-w-[220px]">
   322→                        <Mail className="w-3 h-3 shrink-0" /> {u.email}
   323→                      </span>
   324→                    )}
   325→                    {u.phone && (
   326→                      <span className="inline-flex items-center gap-1">
   327→                        <Phone className="w-3 h-3" /> {u.phone}
   328→                      </span>
   329→                    )}
   330→                    {u.gender && (
   331→                      <span className="inline-flex items-center rounded-md bg-[#FBF9F7] border border-[#E7E5E4] px-1.5 py-0.5 text-[10px] uppercase tracking-[0.14em] text-stone-500">
   332→                        {u.gender}
   333→                      </span>
   334→                    )}
   335→                    {!u.email && !u.phone && !u.gender && (
   336→                      <span className="italic text-stone-400">
   337→                        No contact details
   338→                      </span>
   339→                    )}
   340→                  </div>
   341→                </div>
   342→                <div className="flex items-center gap-2 shrink-0">
   343→                  <StatusBadge status={roleLabel(u.role)} />
   344→                  {isOwner && (
   345→                    <span className="hidden sm:inline text-[10px] uppercase tracking-[0.18em] text-stone-300">
   346→                      Permanent
   347→                    </span>
   348→                  )}
   349→                </div>
   350→              </div>
   351→            );
   352→          })}
   353→        </div>
   354→      )}
   355→    </SectionCard>
   356→  );

   359→/* ========================== Freelancers card ========================= */
   360→
   361→function FreelancersCard({
   362→  freelancers,
   363→  isLoading,
   364→  onAdd,
   365→}: {
   366→  freelancers: Freelancer[];
   367→  isLoading: boolean;
   368→  onAdd: () => void;
   369→}) {
   370→  const qc = useQueryClient();
   371→  const [deleteId, setDeleteId] = useState<string | null>(null);
   372→
   373→  const statusMutation = useMutation({
   374→    mutationFn: ({ id, status }: { id: string; status: string }) =>
   375→      apiPatch<Freelancer>(`/api/freelancers/${id}`, { status }),
   376→    onSuccess: (f) => {
   377→      qc.invalidateQueries({ queryKey: ["freelancers"] });
   378→      toast.success(`${f.name} is now ${f.status}`);
   379→    },
   380→    onError: () => toast.error("Could not update status."),
   381→  });
   382→
   383→  const deleteMutation = useMutation({
   384→    mutationFn: (id: string) => apiDelete(`/api/freelancers/${id}`),
   385→    onSuccess: () => {
   386→      qc.invalidateQueries({ queryKey: ["freelancers"] });
   387→      toast.success("Freelancer removed from roster.");
   388→      setDeleteId(null);
   389→    },
   390→    onError: () => toast.error("Could not remove freelancer."),
   391→  });
   392→
   393→  const target = freelancers.find((f) => f.id === deleteId) || null;
   394→
   395→  return (
   396→    <SectionCard
   397→      title={`Commissioned Talent · ${freelancers.length}`}
   398→      bodyClassName="p-0"
   399→      action={
   400→        <PremiumButton size="sm" onClick={onAdd}>
   401→          <Plus className="w-4 h-4" /> Add Freelancer
   402→        </PremiumButton>
   403→      }
   404→    >
   405→      {isLoading ? (
   406→        <SkeletonList rows={4} />
   407→      ) : freelancers.length === 0 ? (
   408→        <EmptyState
   409→          icon={UsersRound}
   410→          title="No freelancers yet"
   411→          description="Add the beaders, embroiderers, cutters and tailors you commission for bespoke work."
   412→          action={
   413→            <PremiumButton size="sm" onClick={onAdd}>
   414→              <Plus className="w-4 h-4" /> Add Freelancer
   415→            </PremiumButton>
   416→          }
   417→        />
   418→      ) : (
   419→        <div className="divide-y divide-[#E7E5E4]/70">
   420→          {freelancers.map((f) => {
   421→            const SpecialtyIcon =
   422→              SPECIALTY_ICON[f.specialty || "Other"] || Briefcase;
   423→            return (
   424→              <div
   425→                key={f.id}
   426→                className="group flex flex-col sm:flex-row sm:items-center gap-3 sm:gap-4 px-5 py-4 hover:bg-[#faf8f5] transition-colors"
   427→              >
   428→                <div
   429→                  className="w-11 h-11 rounded-full bg-gradient-to-br from-[#800020] to-[#6b0018] flex items-center justify-center text-white shrink-0 shadow-sm"
   430→                  aria-hidden
   431→                >
   432→                  <SpecialtyIcon className="w-5 h-5" />
   433→                </div>
   434→                <div className="flex-1 min-w-0">
   435→                  <div className="flex items-center gap-2 flex-wrap">
   436→                    <span className="font-medium text-[#1C1917] truncate">
   437→                      {f.name}
   438→                    </span>
   439→                    {f.specialty && (
   440→                      <span className="inline-flex items-center rounded-md bg-[#800020]/8 text-[#800020] px-1.5 py-0.5 text-[10px] font-semibold uppercase tracking-[0.14em] border border-[#800020]/15">
   441→                        {f.specialty}
   442→                      </span>
   443→                    )}
   444→                  </div>
   445→                  <div className="flex flex-wrap items-center gap-x-4 gap-y-1 text-xs text-stone-500 mt-1">
   446→                    {f.skills && (
   447→                      <span className="italic text-stone-400 truncate max-w-[280px]">
   448→                        {f.skills}
   449→                      </span>
   450→                    )}
   451→                    {f.phone && (
   452→                      <span className="inline-flex items-center gap-1">
   453→                        <Phone className="w-3 h-3" /> {f.phone}
   454→                      </span>
   455→                    )}
   456→                    {f.email && (
   457→                      <span className="inline-flex items-center gap-1 truncate max-w-[200px]">
   458→                        <Mail className="w-3 h-3 shrink-0" /> {f.email}
   459→                      </span>
   460→                    )}
   461→                    {!f.skills && !f.phone && !f.email && (
   462→                      <span className="italic text-stone-400">
   463→                        No contact details
   464→                      </span>
   465→                    )}
   466→                  </div>
   467→                </div>

   483→                  >
   484→                    <SelectTrigger
   485→                      size="sm"
   486→                      className="w-[130px] bg-white border-[#E7E5E4] rounded-md h-9"
   487→                      disabled={statusMutation.isPending}
   488→                      aria-label={`Change status for ${f.name}`}

   490→                      <SelectValue />
   491→                    </SelectTrigger>
   492→                    <SelectContent>

   498→                    </SelectContent>
   499→                  </Select>
   500→                  <PremiumButton
   501→                    variant="ghost"
   502→                    size="icon"
   503→                    onClick={() => setDeleteId(f.id)}
   504→                    aria-label={`Remove ${f.name}`}
   505→                    className="text-stone-400 hover:text-[#b91c1c] hover:bg-[#b91c1c]/8"

   507→                    <Trash2 className="w-4 h-4" />
   508→                  </PremiumButton>
   509→                </div>
   510→              </div>
   511→            );
   512→          })}
   513→        </div>
   514→      )}
   515→
   516→      <AlertDialog open={!!deleteId} onOpenChange={(o) => !o && setDeleteId(null)}>
   517→        <AlertDialogContent>
   518→          <AlertDialogHeader>
   519→            <AlertDialogTitle>Remove freelancer?</AlertDialogTitle>
   520→            <AlertDialogDescription>
   521→              {target
   522→                ? `This will permanently remove ${target.name} from your roster. This action cannot be undone.`
   523→                : "This will permanently remove the freelancer from your roster. This action cannot be undone."}
   524→            </AlertDialogDescription>
   525→          </AlertDialogHeader>
   526→          <AlertDialogFooter>
   527→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   528→            <AlertDialogAction
   529→              className="bg-[#b91c1c] hover:bg-[#991b1b] text-white border-[#b91c1c]"
   530→              disabled={deleteMutation.isPending}
   531→              onClick={() => deleteId && deleteMutation.mutate(deleteId)}
   532→            >
   533→              {deleteMutation.isPending ? "Removing…" : "Remove"}
   534→            </AlertDialogAction>
   535→          </AlertDialogFooter>
   536→        </AlertDialogContent>
   537→      </AlertDialog>
   538→    </SectionCard>
   539→  );
   540→}

   557→  gender: "",

   561→
   562→function AddMemberDialog({
   563→  open,
   564→  onOpenChange,
   565→  onDone,
   566→}: {
   567→  open: boolean;
   568→  onOpenChange: (v: boolean) => void;
   569→  onDone: () => void;
   570→}) {
   571→  const [form, setForm] = useState<MemberForm>(EMPTY_MEMBER);

   573→  const createMutation = useMutation({
   574→    mutationFn: (body: MemberForm) => apiPost<User>("/api/users", body),
   575→    onSuccess: (u) => {

   577→        description: `Signed in as ${u.role}. Default password: ${form.password || "atelier2025"}.`,
