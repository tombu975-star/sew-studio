   113→   113→        status: "Scheduled",

   142→   142→  // Notes mutation

   145→   145→      apiPatch<Order>(`/api/orders/${selectedId}`, { notes }),
   146→   146→    onSuccess: async (updated) => {
   147→   147→      toast.success("Notes saved");
   148→   148→      qc.setQueryData(["order", selectedId], updated);
   149→   149→      qc.invalidateQueries({ queryKey: ["orders"] });
   150→   150→    },
   151→   151→    onError: (e: Error) => toast.error(e.message),

   156→   156→      <div className="p-5 md:p-8 max-w-[1200px] mx-auto w-full">

   171→   171→          <div className="py-10 text-center text-sm text-stone-400">
   172→   172→            Loading order…

   180→   180→
   181→   181→  return (
   182→   182→    <div className="p-5 md:p-8 max-w-[1200px] mx-auto w-full">
   183→   183→      <button
   184→   184→        onClick={() => go("orders")}
   185→   185→        className="inline-flex items-center gap-1.5 text-xs text-stone-500 hover:text-[#800020] transition mb-4"

   239→   239→          <Select

   287→   287→              />

   330→   330→          {/* Fittings */}

   480→   480→/* ---------- Helpers ---------- */
   481→   481→
   482→   482→function Stat({
