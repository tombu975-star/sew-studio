    72→// ---------------------------------------------------------------------------
    73→// Grade helpers
    74→// ---------------------------------------------------------------------------
    75→
    76→function scorePercent(score?: number | null, max?: number | null): number | null {
    77→  if (score == null || max == null || max === 0) return null;
    78→  return (score / max) * 100;
    79→}
    80→
    81→function letterGrade(pct: number): string {
    82→  if (pct >= 90) return "A";
    83→  if (pct >= 80) return "B";
    84→  if (pct >= 70) return "C";
    85→  if (pct >= 60) return "D";
    86→  return "F";
    87→}
    88→
    89→function pctTone(pct: number | null): "emerald" | "amber" | "orange" | "muted" {
    90→  if (pct == null) return "muted";
    91→  if (pct >= 80) return "emerald";
    92→  if (pct >= 60) return "amber";
    93→  return "orange";
    94→}
    95→
    96→const TONE_BADGE: Record<string, string> = {
    97→  emerald: "bg-emerald-100 text-emerald-800 border-emerald-200",
    98→  amber: "bg-amber-100 text-amber-800 border-amber-200",
    99→  orange: "bg-orange-100 text-orange-800 border-orange-200",
   100→  muted: "bg-stone-100 text-stone-600 border-stone-200",
   101→};
   102→
   103→const TONE_DOT: Record<string, string> = {
   104→  emerald: "bg-emerald-600",
   105→  amber: "bg-amber-600",
   106→  orange: "bg-orange-600",
   107→  muted: "bg-stone-400",
   108→};
   109→
   110→/** Small pill showing "18/20" colored by percentage. */
   111→function ScoreBadge({
   112→  score,
   113→  max,
   114→  className,
   115→}: {
   116→  score: number;
   117→  max: number;
   118→  className?: string;
   119→}) {
   120→  const pct = scorePercent(score, max);
   121→  const tone = pctTone(pct);
   122→  return (
   123→    <span
   124→      className={cn(
   125→        "inline-flex items-center gap-1.5 px-2 py-0.5 text-[11px] font-semibold border rounded-md tabular-nums",
   126→        TONE_BADGE[tone],
   127→        className
   128→      )}
   129→    >
   130→      <Star className="w-3 h-3" />
   131→      {score}/{max}
   132→    </span>
   133→  );
   134→}
   135→
   136→/** Letter grade pill (A/B/C/D/F) colored by percentage. */
   137→function LetterGradeBadge({ pct }: { pct: number }) {
   138→  const tone = pctTone(pct);
   139→  return (
   140→    <span
   141→      className={cn(
   142→        "inline-flex items-center gap-1.5 px-2 py-0.5 text-[11px] font-semibold uppercase tracking-wide border rounded-md",
   143→        TONE_BADGE[tone]
   144→      )}
   145→    >
   146→      <span className={cn("w-1.5 h-1.5 rounded-full", TONE_DOT[tone])} />
   147→      {letterGrade(pct)}
   148→    </span>
   149→  );
   150→}

  1002→function GradeForm({
  1003→  task,
  1004→  onSubmit,
  1005→  onCancel,
  1006→  submitting,
  1007→}: {

  1009→  onSubmit: (payload: GradePayload) => void;

  1011→  submitting: boolean;
  1012→}) {
  1013→  // Lazy initializers — fields prefilled from the task (incl. already-graded tasks).
  1014→  // The parent uses `key={task.id}` so this component remounts per task.
  1015→  const [maxScore, setMaxScore] = useState<number>(
  1016→    () => (task.maxScore != null ? task.maxScore : 20)
  1017→  );
  1018→  const [score, setScore] = useState<number>(() => task.score ?? 0);
  1019→  const [feedback, setFeedback] = useState<string>(() => task.feedback ?? "");
  1020→
  1021→  const safeMax = maxScore > 0 ? maxScore : 0;
  1022→  const pct = safeMax > 0 ? (score / safeMax) * 100 : 0;

  1024→    safeMax > 0 && Number.isFinite(score) && score >= 0 && score <= safeMax;

  1033→      toast.error(`Score must be between 0 and ${safeMax}`);

  1035→    }

  1037→      maxScore: safeMax,
  1038→      score,
  1039→      feedback: feedback.trim() || undefined,
