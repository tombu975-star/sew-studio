    82→function orderPaidTotal(orderId: string, payments: Payment[]) {
    83→  return payments
    84→    .filter((p) => p.orderId === orderId)
    85→    .reduce((s, p) => s + (Number(p.amount) || 0), 0);
    86→}

    88→function orderCostTotal(order: Order) {
    89→  return (Number(order.fabricCost) || 0) + (Number(order.labourCost) || 0);
    90→}

    92→function sameMonth(d: string | Date | null | undefined, ref: Date) {
    93→  if (!d) return false;
    94→  const x = new Date(d);
    95→  if (isNaN(x.getTime())) return false;
    96→  return x.getMonth() === ref.getMonth() && x.getFullYear() === ref.getFullYear();
    97→}

    99→/** Build last `months` monthly buckets from a flat array using extractor fns. */
   100→function monthlyBuckets<T>(
   101→  items: T[],
   102→  getDate: (x: T) => string | Date | null | undefined,
   103→  getValue: (x: T) => number,
   104→  months = 6
   105→) {
   106→  const now = new Date();
   107→  const buckets: { key: string; label: string; total: number }[] = [];
   108→  for (let i = months - 1; i >= 0; i--) {
   109→    const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
   110→    buckets.push({
   111→      key: `${d.getFullYear()}-${d.getMonth()}`,
   112→      label: d.toLocaleDateString("en-GB", { month: "short" }),
   113→      total: 0,
   114→    });
   115→  }
   116→  for (const item of items) {
   117→    const dRaw = getDate(item);
   118→    if (!dRaw) continue;
   119→    const d = new Date(dRaw);
   120→    if (isNaN(d.getTime())) continue;
   121→    const key = `${d.getFullYear()}-${d.getMonth()}`;
   122→    const b = buckets.find((x) => x.key === key);
   123→    if (b) b.total += Number(getValue(item)) || 0;
   124→  }
   125→  return buckets;
   126→}

   155→function avatarGradient(seed: string) {
   156→  let hash = 0;
   157→  for (let i = 0; i < seed.length; i++)
   158→    hash = (hash * 31 + seed.charCodeAt(i)) | 0;
   159→  return AVATAR_GRADIENTS[Math.abs(hash) % AVATAR_GRADIENTS.length];
   160→}
