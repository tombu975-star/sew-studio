   107→   107→function gradientFor(name: string): string {
   108→   108→  // Deterministic burgundy-leaning gradient avatar.
   109→   109→  const palettes = [
   110→   110→    "from-[#800020] to-[#6b0018]",
   111→   111→    "from-[#800020] to-[#b45309]",
   112→   112→    "from-[#6b0018] to-[#0e7490]",
   113→   113→    "from-[#800020] to-[#15803D]",
   114→   114→    "from-[#6b0018] to-[#800020]",
   115→   115→  ];
   116→   116→  let h = 0;
   117→   117→  for (let i = 0; i < name.length; i++) h = (h * 31 + name.charCodeAt(i)) >>> 0;
   118→   118→  return palettes[h % palettes.length];
   119→   119→}

   390→   390→/* ===================== Shared progress body ===================== */
   391→   391→
   392→   392→function ProgressBody({
   393→   393→  traineeId,
   394→   394→  viewerRole,
   395→   395→  mentorId,
   396→   396→}: {
   397→   397→  traineeId: string;
   398→   398→  viewerRole: "trainee" | "owner";
   399→   399→  mentorId?: string;
   400→   400→}) {
   401→   401→  const qc = useQueryClient();
   402→   402→  const [logOpen, setLogOpen] = useState(false);
   403→   403→
   404→   404→  // Training logs for this trainee.
   405→   405→  const logsQ = useQuery<TrainingLog[]>({
   406→   406→    queryKey: ["training-logs", traineeId],
   407→   407→    queryFn: () =>
   408→   408→      apiGet<TrainingLog[]>(`/api/training-logs?traineeId=${traineeId}`),
   409→   409→  });
   410→   410→
   411→   411→  // Tasks for this trainee.
   412→   412→  const tasksQ = useQuery<Task[]>({
   413→   413→    queryKey: ["tasks"],
   414→   414→    queryFn: () => apiGet<Task[]>("/api/tasks"),
   415→   415→  });
   416→   416→
   417→   417→  // Orders for this trainee (assignedStaffId).
   418→   418→  const ordersQ = useQuery<Order[]>({
   419→   419→    queryKey: ["orders"],
   420→   420→    queryFn: () => apiGet<Order[]>("/api/orders"),
   421→   421→  });
   422→   422→
   423→   423→  const logs = logsQ.data ?? [];
   424→   424→
   425→   425→  const stats = useMemo(() => {
   426→   426→    const total = logs.length;
   427→   427→    const proficient = logs.filter((l) => l.status === "Proficient").length;
   428→   428→    const practicing = logs.filter((l) => l.status === "Practicing").length;
   429→   429→    const learning = logs.filter((l) => l.status === "Learning").length;
   430→   430→    return { total, proficient, practicing, learning };
   431→   431→  }, [logs]);
   432→   432→
   433→   433→  // Group logs by skill for the mastery overview.
   434→   434→  const skillGroups = useMemo(() => {
   435→   435→    const map = new Map<
   436→   436→      string,
   437→   437→      { skill: string; logs: TrainingLog[]; latest: TrainingLog }
   438→   438→    >();
   439→   439→    for (const log of logs) {
   440→   440→      const skill = log.skill?.trim() || "General";
   441→   441→      const existing = map.get(skill);
   442→   442→      if (!existing) {
   443→   443→        map.set(skill, { skill, logs: [log], latest: log });
   444→   444→      } else {
   445→   445→        existing.logs.push(log);
   446→   446→        // logs come newest-first from the API (date desc); keep latest as the first seen
   447→   447→        // (which is the most recent by date).
   448→   448→        if (
   449→   449→          new Date(log.date).getTime() > new Date(existing.latest.date).getTime()
   450→   450→        ) {
   451→   451→          existing.latest = log;
   452→   452→        }
   453→   453→      }
   454→   454→    }
   455→   455→    // Sort skills by latest entry date desc.
   456→   456→    return Array.from(map.values()).sort(
   457→   457→      (a, b) =>
   458→   458→        new Date(b.latest.date).getTime() - new Date(a.latest.date).getTime()
   459→   459→    );
   460→   460→  }, [logs]);
   461→   461→
   462→   462→  const myTasks = useMemo(
   463→   463→    () =>
   464→   464→      (tasksQ.data ?? [])
   465→   465→        .filter((t) => t.staffId === traineeId)
   466→   466→        .sort((a, b) => {
   467→   467→          // Pending first, then by due date asc.
   468→   468→          const aDone = a.status === "Done" ? 1 : 0;
   469→   469→          const bDone = b.status === "Done" ? 1 : 0;
   470→   470→          if (aDone !== bDone) return aDone - bDone;
   471→   471→          const ad = a.dueDate ? new Date(a.dueDate).getTime() : Infinity;
   472→   472→          const bd = b.dueDate ? new Date(b.dueDate).getTime() : Infinity;
   473→   473→          return ad - bd;
   474→   474→        }),
   475→   475→    [tasksQ.data, traineeId]
   476→   476→  );
   477→   477→
   478→   478→  const myOrders = useMemo(
   479→   479→    () =>
   480→   480→      (ordersQ.data ?? [])
   481→   481→        .filter((o) => o.assignedStaffId === traineeId)
   482→   482→        .sort(
   483→   483→          (a, b) =>
   484→   484→            new Date(b.orderDate).getTime() - new Date(a.orderDate).getTime()
   485→   485→        )
   486→   486→        .slice(0, 8),
   487→   487→    [ordersQ.data, traineeId]
   488→   488→  );
   489→   489→
   490→   490→  const invalidateAll = () => {
   491→   491→    qc.invalidateQueries({ queryKey: ["training-logs", traineeId] });
   492→   492→    qc.invalidateQueries({ queryKey: ["training-logs"] });
   493→   493→  };
   494→   494→
   495→   495→  return (
   496→   496→    <>
   497→   497→      {/* Premium stat strip */}
   498→   498→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4 mb-6">
   499→   499→        <PremiumMetric
   500→   500→          label="Total Logs"
