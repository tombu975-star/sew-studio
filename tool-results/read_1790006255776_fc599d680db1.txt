   107→function gradientFor(name: string): string {
   108→  // Deterministic burgundy-leaning gradient avatar.
   109→  const palettes = [
   110→    "from-[#800020] to-[#6b0018]",
   111→    "from-[#800020] to-[#b45309]",
   112→    "from-[#6b0018] to-[#0e7490]",
   113→    "from-[#800020] to-[#15803D]",
   114→    "from-[#6b0018] to-[#800020]",
   115→  ];
   116→  let h = 0;
   117→  for (let i = 0; i < name.length; i++) h = (h * 31 + name.charCodeAt(i)) >>> 0;
   118→  return palettes[h % palettes.length];
   119→}

   390→/* ===================== Shared progress body ===================== */
   391→
   392→function ProgressBody({
   393→  traineeId,
   394→  viewerRole,
   395→  mentorId,
   396→}: {
   397→  traineeId: string;
   398→  viewerRole: "trainee" | "owner";
   399→  mentorId?: string;
   400→}) {
   401→  const qc = useQueryClient();
   402→  const [logOpen, setLogOpen] = useState(false);
   403→
   404→  // Training logs for this trainee.
   405→  const logsQ = useQuery<TrainingLog[]>({
   406→    queryKey: ["training-logs", traineeId],
   407→    queryFn: () =>
   408→      apiGet<TrainingLog[]>(`/api/training-logs?traineeId=${traineeId}`),
   409→  });
   410→
   411→  // Tasks for this trainee.
   412→  const tasksQ = useQuery<Task[]>({
   413→    queryKey: ["tasks"],
   414→    queryFn: () => apiGet<Task[]>("/api/tasks"),
   415→  });
   416→
   417→  // Orders for this trainee (assignedStaffId).
   418→  const ordersQ = useQuery<Order[]>({
   419→    queryKey: ["orders"],
   420→    queryFn: () => apiGet<Order[]>("/api/orders"),
   421→  });
   422→
   423→  const logs = logsQ.data ?? [];
   424→
   425→  const stats = useMemo(() => {
   426→    const total = logs.length;
   427→    const proficient = logs.filter((l) => l.status === "Proficient").length;
   428→    const practicing = logs.filter((l) => l.status === "Practicing").length;
   429→    const learning = logs.filter((l) => l.status === "Learning").length;
   430→    return { total, proficient, practicing, learning };
   431→  }, [logs]);
   432→
   433→  // Group logs by skill for the mastery overview.
   434→  const skillGroups = useMemo(() => {
   435→    const map = new Map<
   436→      string,
   437→      { skill: string; logs: TrainingLog[]; latest: TrainingLog }
   438→    >();
   439→    for (const log of logs) {
   440→      const skill = log.skill?.trim() || "General";
   441→      const existing = map.get(skill);
   442→      if (!existing) {
   443→        map.set(skill, { skill, logs: [log], latest: log });
   444→      } else {
   445→        existing.logs.push(log);
   446→        // logs come newest-first from the API (date desc); keep latest as the first seen
   447→        // (which is the most recent by date).
   448→        if (
   449→          new Date(log.date).getTime() > new Date(existing.latest.date).getTime()
   450→        ) {
   451→          existing.latest = log;
   452→        }
   453→      }
   454→    }
   455→    // Sort skills by latest entry date desc.
   456→    return Array.from(map.values()).sort(
   457→      (a, b) =>
   458→        new Date(b.latest.date).getTime() - new Date(a.latest.date).getTime()
   459→    );
   460→  }, [logs]);
   461→
   462→  const myTasks = useMemo(
   463→    () =>
   464→      (tasksQ.data ?? [])
   465→        .filter((t) => t.staffId === traineeId)
   466→        .sort((a, b) => {
   467→          // Pending first, then by due date asc.
   468→          const aDone = a.status === "Done" ? 1 : 0;
   469→          const bDone = b.status === "Done" ? 1 : 0;
   470→          if (aDone !== bDone) return aDone - bDone;
   471→          const ad = a.dueDate ? new Date(a.dueDate).getTime() : Infinity;
   472→          const bd = b.dueDate ? new Date(b.dueDate).getTime() : Infinity;
   473→          return ad - bd;
   474→        }),
   475→    [tasksQ.data, traineeId]
   476→  );
   477→
   478→  const myOrders = useMemo(
   479→    () =>
   480→      (ordersQ.data ?? [])
   481→        .filter((o) => o.assignedStaffId === traineeId)
   482→        .sort(
   483→          (a, b) =>
   484→            new Date(b.orderDate).getTime() - new Date(a.orderDate).getTime()
   485→        )
   486→        .slice(0, 8),
   487→    [ordersQ.data, traineeId]
   488→  );
   489→
   490→  const invalidateAll = () => {
   491→    qc.invalidateQueries({ queryKey: ["training-logs", traineeId] });
   492→    qc.invalidateQueries({ queryKey: ["training-logs"] });
   493→  };
   494→
   495→  return (
   496→    <>
   497→      {/* Premium stat strip */}
   498→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4 mb-6">
   499→        <PremiumMetric
   500→          label="Total Logs"
   501→          value={<AnimatedNumber value={stats.total} />}
   502→          icon={GraduationCap}
   503→          accent
   504→        />
   505→        <PremiumMetric
   506→          label="Proficient"
   507→          value={<AnimatedNumber value={stats.proficient} />}
   508→          icon={Sparkles}
   509→        />
   510→        <PremiumMetric
   511→          label="Practicing"
   512→          value={<AnimatedNumber value={stats.practicing} />}
   513→          icon={Scissors}
   514→        />
   515→        <PremiumMetric
   516→          label="Learning"
   517→          value={<AnimatedNumber value={stats.learning} />}
   518→          icon={GraduationCap}
   519→        />
   520→      </div>
   521→
   522→      {/* Skills mastery overview */}
   523→      <div className="mb-6">
   524→        <SkillMasteryCard
   525→          skillGroups={skillGroups}
   526→          isLoading={logsQ.isLoading}
   527→        />
   528→      </div>
   529→
   530→      {/* Training log timeline */}
   531→      <div className="mb-6">
   532→        <SectionCard
   533→          title="Training log"
   534→          bodyClassName="p-0"
   535→          action={
   536→            <PremiumButton size="sm" onClick={() => setLogOpen(true)}>
   537→              <Plus className="w-4 h-4" />
   538→              {viewerRole === "trainee" ? "Log new learning" : "Log for trainee"}
   539→            </PremiumButton>
   540→          }
   541→        >
   542→          {logsQ.isLoading ? (
   543→            <SkeletonList rows={4} />
   544→          ) : logs.length === 0 ? (
   545→            <EmptyState
   546→              icon={GraduationCap}
   547→              title="No log entries yet"
   548→              description={
   549→                viewerRole === "trainee"
   550→                  ? "Log your first learning to start tracking your skill journey."
   551→                  : "Log the first learning entry for this trainee."
   552→              }
   553→              action={
   554→                <PremiumButton size="sm" onClick={() => setLogOpen(true)}>
   555→                  <Plus className="w-4 h-4" />{" "}
   556→                  {viewerRole === "trainee"
   557→                    ? "Log new learning"
   558→                    : "Log for trainee"}
   559→                </PremiumButton>
   560→              }
   561→            />
   562→          ) : (
   563→            <div className="divide-y divide-[#E7E5E4]/70">
   564→              {logs.map((log, idx) => (
   565→                <TrainingLogRow
   566→                  key={log.id}
   567→                  log={log}
   568→                  index={idx}
   569→                  canEdit
   570→                  onInvalidated={invalidateAll}
   571→                />
   572→              ))}
   573→            </div>
   574→          )}
   575→        </SectionCard>
   576→      </div>
   577→
   578→      {/* My tasks */}
   579→      <div className="mb-6">
   580→        <MyTasksCard tasks={myTasks} isLoading={tasksQ.isLoading} />
   581→      </div>
   582→
   583→      {/* Orders contributed */}
   584→      <div className="mb-2">
   585→        <OrdersContributedCard orders={myOrders} isLoading={ordersQ.isLoading} />
   586→      </div>
   587→
   588→      {/* Log dialog */}
   589→      <LogDialog
   590→        open={logOpen}
   591→        onOpenChange={setLogOpen}
   592→        traineeId={traineeId}
   593→        mentorId={viewerRole === "owner" ? mentorId : undefined}
   594→        onDone={invalidateAll}
   595→      />
   596→    </>
   597→  );
   598→}

   600→/* ======================= Skill mastery card ======================= */
   601→
   602→function SkillMasteryCard({
   603→  skillGroups,
   604→  isLoading,
   605→}: {
   606→  skillGroups: {
   607→    skill: string;
   608→    logs: TrainingLog[];
   609→    latest: TrainingLog;
   610→  }[];
   611→  isLoading: boolean;
   612→}) {
   613→  return (
   614→    <SectionCard title="Skill mastery" bodyClassName="p-0">
   615→      {isLoading ? (
   616→        <SkeletonList rows={3} />
   617→      ) : skillGroups.length === 0 ? (
   618→        <EmptyState
   619→          icon={GraduationCap}
   620→          title="No skills logged yet"
   621→          description="Once training entries are logged, this overview groups them by skill."
   622→        />
   623→      ) : (
   624→        <ul className="divide-y divide-[#E7E5E4]/70">
   625→          {skillGroups.map((group, idx) => {
   626→            const proficientCount = group.logs.filter(
