   480→/* ---------- Helpers ---------- */
   481→
   482→function Stat({
   483→  label,
   484→  value,
   485→  primary,
   486→  tone,
   487→}: {
   488→  label: string;
   489→  value: string;
   490→  primary?: boolean;
   491→  tone?: "ok" | "warn";
   492→}) {
   493→  return (
   494→    <div>
   495→      <div className="text-[10px] uppercase tracking-[0.18em] text-stone-500 font-semibold">
   496→        {label}
   497→      </div>
   498→      <div
   499→        className={cn(
   500→          "mt-1 text-lg font-display font-semibold",
   501→          primary && "text-[#1C1917]",
   502→          tone === "ok" && "text-emerald-700",
   503→          tone === "warn" && "text-amber-700",
   504→          !primary && !tone && "text-[#1C1917]"
   505→        )}
   506→      >
   507→        {value}
   508→      </div>
   509→    </div>
   510→  );
   511→}
   512→
   513→function Row({ label, value }: { label: string; value?: string | null }) {
   514→  if (!value) return null;
   515→  return (
   516→    <div className="flex justify-between gap-3">
   517→      <span className="text-stone-400">{label}</span>
   518→      <span className="text-[#1C1917] text-right">{value}</span>
   519→    </div>
   520→  );
   521→}
   522→
   523→function NotesEditor({
   524→  initial,
   525→  onSave,
   526→  saving,
   527→}: {
   528→  initial: string;
   529→  onSave: (v: string) => void;
   530→  saving: boolean;
   531→}) {
   532→  const [val, setVal] = useState(initial);
   533→  const [editing, setEditing] = useState(false);
   534→  const [lastInitial, setLastInitial] = useState(initial);
   535→  if (initial !== lastInitial) {
   536→    setLastInitial(initial);
   537→    setVal(initial);
   538→  }
   539→
   540→  if (!editing) {
   541→    return (
   542→      <div>
   543→        {initial ? (
   544→          <p className="text-sm text-stone-700 whitespace-pre-wrap">
   545→            {initial}
   546→          </p>
   547→        ) : (
   548→          <p className="text-sm text-stone-400">No notes yet.</p>
   549→        )}
   550→        <div className="mt-3">
   551→          <Button
   552→            variant="outline"
   553→            size="sm"
   554→            className="rounded-sm"
   555→            onClick={() => setEditing(true)}
   556→          >
   557→            <StickyNote className="w-3.5 h-3.5" /> Edit notes
   558→          </Button>
   559→        </div>
   560→      </div>
   561→    );
   562→  }
   563→  return (
   564→    <div>
   565→      <Textarea
   566→        value={val}
   567→        onChange={(e) => setVal(e.target.value)}
   568→        rows={5}
   569→        placeholder="Notes about this order…"
   570→        className="rounded-sm"
   571→      />
   572→      <div className="flex gap-2 mt-3">
   573→        <Button
   574→          size="sm"
   575→          className="bg-[#800020] hover:bg-[#6b0018] text-white rounded-sm"
   576→          disabled={saving}
   577→          onClick={() => {
   578→            onSave(val);
   579→            setEditing(false);
   580→          }}
   581→        >
   582→          {saving ? "Saving…" : "Save"}
   583→        </Button>
   584→        <Button
   585→          variant="outline"
   586→          size="sm"
   587→          className="rounded-sm"
   588→          onClick={() => {
   589→            setVal(initial);
   590→            setEditing(false);
   591→          }}
   592→        >
   593→          Cancel
   594→        </Button>
   595→      </div>
   596→    </div>
   597→  );
   598→}

   602→function PaymentDialog({
   603→  onConfirm,
   604→}: {
   605→  onConfirm: (b: { amount: number; method?: string; note?: string }) => void;
   606→}) {
